cmake_minimum_required(VERSION 3.14)
project(slice_smoke LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Dyninst REQUIRED COMPONENTS parseAPI instructionAPI symtabAPI common)
add_executable(slice_smoke slice_smoke.C)
target_link_libraries(slice_smoke PRIVATE Dyninst::parseAPI Dyninst::instructionAPI Dyninst::symtabAPI Dyninst::common)
add_executable(depredicate depredicate.C)
target_link_libraries(depredicate PRIVATE Dyninst::parseAPI Dyninst::instructionAPI Dyninst::symtabAPI Dyninst::common)
add_executable(bracket bracket.C)
target_link_libraries(bracket PRIVATE Dyninst::parseAPI Dyninst::instructionAPI Dyninst::symtabAPI Dyninst::common)
add_executable(classify classify.C)
target_link_libraries(classify PRIVATE Dyninst::parseAPI Dyninst::instructionAPI Dyninst::symtabAPI Dyninst::common)
add_executable(structure structure.C)
target_link_libraries(structure PRIVATE Dyninst::parseAPI Dyninst::instructionAPI Dyninst::symtabAPI Dyninst::common)
add_executable(cfgraph cfgraph.C)
target_link_libraries(cfgraph PRIVATE Dyninst::parseAPI Dyninst::instructionAPI Dyninst::symtabAPI Dyninst::common)
